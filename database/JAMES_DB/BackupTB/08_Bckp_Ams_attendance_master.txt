CREATE TABLE Bckp_Ams_attendance_master(att_no BIGINT AUTO_INCREMENT,spid VARCHAR(10) NOT NULL REFERENCES Bckp_Students(spid) ON UPDATE CASCADE,ams_setup_id  INTEGER NOT NULL REFERENCES Bckp_Ams_setup_course_subject_map(ams_setup_id) ON UPDATE CASCADE,fid VARCHAR(10) NOT NULL REFERENCES Faculties(fid) ON UPDATE CASCADE,att_date_time DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,att_status BOOLEAN NOT NULL,PRIMARY KEY(att_no));


CREATE INDEX idx_AmsAttMasterAmsSetupSpid_bckp ON Bckp_Ams_attendance_master(spid); 
CREATE INDEX idx_AmsAttMasterAmsSetupId_bckp ON Bckp_Ams_attendance_master(ams_setup_id);
