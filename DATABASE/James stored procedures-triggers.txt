DELIMITER $$
CREATE OR REPLACE TRIGGER insert_ams_api BEFORE INSERT ON Ams_api
FOR EACH ROW
BEGIN
DECLARE sem int;
SELECT cur_semester into sem  FROM vw_students where spid = NEW.spid;
SET NEW.semester := sem;
END $$
DELIMITER ;
